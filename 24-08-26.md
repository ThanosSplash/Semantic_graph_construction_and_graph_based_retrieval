## Προτεινόμενο refactor

```python
def sim_scores_for_query(
    query_id,
    query_store,
    corpus_store,
    retrieved_ids=None
):
    query = query_store[query_id]
    query_emb = query[0].reshape(1, -1)
    correct_results = query[1]

    if retrieved_ids is None:
        documents = list(corpus_store.keys())
    else:
        documents = [
            doc_id
            for doc_id in retrieved_ids
            if doc_id in corpus_store
        ]

    embeddings = np.vstack([
        corpus_store[doc_id]
        for doc_id in documents
    ])

    sims = cosine_similarity(
        query_emb,
        embeddings
    ).flatten()

    sims = np.maximum(sims, 0)

    return dict(zip(documents, sims)), correct_results
```

## Κάνε μόνο coverage audit

```python
def coverage_report(query_store, corpus):
    corpus_ids = set(corpus.keys())

    all_gold = set()
    zero_reachable = 0

    for query_id, query_data in query_store.items():
        gold = set(query_data[1])

        all_gold.update(gold)

        if len(gold & corpus_ids) == 0:
            zero_reachable += 1

    present = all_gold & corpus_ids
    missing = all_gold - corpus_ids

    coverage = (
        len(present) / len(all_gold)
        if all_gold
        else 0.0
    )

    return {
        "queries": len(query_store),
        "total_unique_gold": len(all_gold),
        "gold_present": len(present),
        "gold_missing": len(missing),
        "coverage": coverage,
        "queries_with_zero_reachable_gold": zero_reachable
    }
```

# 2.4 Διόρθωσε το missing rank

Πιο καθαρό:

```python
def RR_score(predictions, correct_results, k):
    correct_set = set(correct_results)

    for idx, prediction in enumerate(predictions[:k]):
        if prediction in correct_set:
            return 1 / (idx + 1), idx + 1

    return 0.0, k + 1
```

Άρα:

```text

rank K      → relevant τελευταίο
```

```text

```

Παράδειγμα:

```python
def get_matching_baseline(
    baseline_queries,
    query_id,
    k
):
    candidates = [
        row
        for row in baseline_queries[query_id]["results"]
        if row["k"] == k
    ]

    if len(candidates) != 1:
        raise ValueError(
            f"Expected exactly one baseline record "
            f"for query={query_id}, k={k}"
        )

    return candidates[0]
```

## 12.1 Paired bootstrap

```python
import numpy as np

def paired_bootstrap_ci(
    baseline,
    graph,
    n_boot=5000,
    seed=42
):
    rng = np.random.default_rng(seed)

    baseline = np.asarray(baseline)
    graph = np.asarray(graph)

    differences = graph - baseline

    means = []

    for _ in range(n_boot):
        sample = rng.choice(
            differences,
            size=len(differences),
            replace=True
        )

        means.append(
            sample.mean()
        )

    return np.percentile(
        means,
        [2.5, 97.5]
    )
```
